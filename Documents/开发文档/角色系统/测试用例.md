public enum RoleType
{
    Player = 0,
    Monster = 1,
    Npc = 2
}

public enum AttributeType
{
    Health,
    Attack,
    Defense,
    Speed
}

public enum RoleEventType
{
    OnDeath,
    OnDamage,
    OnRevive
}

public class RoleEventArgs
{
    public int Damage { get; set; }
    public string Source { get; set; }
}

public class MoveCommand { }
public class AttackCommand { }

public class Role : MonoBehaviour
{
    private Dictionary<RoleEventType, List<System.Action<RoleEventArgs>>> eventHandlers = new Dictionary<RoleEventType, List<System.Action<RoleEventArgs>>>();
    private Dictionary<int, System.Delegate> commandHandlers = new Dictionary<int, System.Delegate>();
    private int commandIdCounter = 0;

    public event System.Action OnInitialized;
    public event System.Action OnUpdated;
    public event System.Action OnDestroyed;

    void Awake()
    {
        gameObject.AddComponent<RoleAttrComponent>();
        gameObject.AddComponent<RoleMoveComponent>();
        gameObject.AddComponent<RoleAttackComponent>();
    }

    void Start()
    {
        OnInitialized?.Invoke();
    }

    void Update()
    {
        OnUpdated?.Invoke();
    }

    void OnDestroy()
    {
        OnDestroyed?.Invoke();
    }

    public void RegisterRoleEvent(RoleEventType eventType, System.Action<RoleEventArgs> handler)
    {
        if (!eventHandlers.ContainsKey(eventType))
            eventHandlers[eventType] = new List<System.Action<RoleEventArgs>>();
        eventHandlers[eventType].Add(handler);
    }

    public void UnregisterRoleEvent(RoleEventType eventType, System.Action<RoleEventArgs> handler)
    {
        if (eventHandlers.ContainsKey(eventType))
            eventHandlers[eventType].Remove(handler);
    }

    public void DispatchRoleEvent(RoleEventType eventType, RoleEventArgs args, bool bubble = false)
    {
        if (eventHandlers.ContainsKey(eventType))
        {
            foreach (var handler in eventHandlers[eventType])
            {
                handler(args);
            }
        }

        if (bubble && transform.parent != null)
        {
            var parentRole = transform.parent.GetComponent<Role>();
            parentRole?.DispatchRoleEvent(eventType, args, bubble);
        }
    }

    public int RegisterCommand<T>(System.Action<T> handler) where T : class
    {
        int id = commandIdCounter++;
        commandHandlers[id] = handler;
        return id;
    }

    public void CancelCommand(int commandId)
    {
        commandHandlers.Remove(commandId);
    }

    public void SendCommand<T>(T command) where T : class
    {
        foreach (var handler in commandHandlers.Values)
        {
            if (handler is System.Action<T> typedHandler)
            {
                typedHandler(command);
            }
        }
    }
}

public class RoleAttrComponent : MonoBehaviour
{
    private Dictionary<AttributeType, float> attributes = new Dictionary<AttributeType, float>();

    public void SetAttribute(AttributeType type, float value)
    {
        attributes[type] = value;
    }

    public float GetAttribute(AttributeType type)
    {
        return attributes.ContainsKey(type) ? attributes[type] : 0;
    }

    public void UpdateAttribute(AttributeType type, float delta)
    {
        if (attributes.ContainsKey(type))
            attributes[type] += delta;
    }

    public void RemoveAttribute(AttributeType type)
    {
        attributes.Remove(type);
    }
}

public class RoleMoveComponent : MonoBehaviour
{
    private Vector3 targetPosition;
    private bool isMoving = false;
    private float moveSpeed = 5f;

    public bool IsMoving => isMoving;

    public void SetTarget(Vector3 target)
    {
        targetPosition = target;
        isMoving = true;
    }

    public void Stop()
    {
        isMoving = false;
    }

    public void Update()
    {
        if (!isMoving) return;

        Vector3 direction = (targetPosition - transform.position).normalized;
        float distance = Vector3.Distance(transform.position, targetPosition);

        if (distance < 0.1f)
        {
            isMoving = false;
            return;
        }

        transform.position += direction * moveSpeed * Time.deltaTime;
    }
}

public class RoleAttackComponent : MonoBehaviour
{
    private float attackRange = 10f;
    private float attackDamage = 10f;

    public bool IsInAttackRange(Vector3 targetPosition)
    {
        return Vector3.Distance(transform.position, targetPosition) <= attackRange;
    }

    public void Attack(Role target)
    {
        if (target == null) return;

        var targetAttr = target.GetComponent<RoleAttrComponent>();
        if (targetAttr == null) return;

        float currentHealth = targetAttr.GetAttribute(AttributeType.Health);
        var attackerAttr = GetComponent<RoleAttrComponent>();
        
        if (attackerAttr != null)
        {
            attackDamage = attackerAttr.GetAttribute(AttributeType.Attack);
        }

        targetAttr.UpdateAttribute(AttributeType.Health, -attackDamage);
        
        var eventArgs = new RoleEventArgs { Damage = (int)attackDamage, Source = gameObject.name };
        target.DispatchRoleEvent(RoleEventType.OnDamage, eventArgs);
    }
}
