using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;
using System.Collections;
using System.Collections.Generic;

namespace CardRougelike.UI.Tests
{
    public class BaseWindowTests
    {
        private TestWindow _testWindow;

        [SetUp]
        public void Setup()
        {
            var gameObject = new GameObject("TestWindow");
            _testWindow = gameObject.AddComponent<TestWindow>();
        }

        [TearDown]
        public void Teardown()
        {
            if (_testWindow != null)
            {
                Object.Destroy(_testWindow.gameObject);
            }
        }

        [Test]
        public void Show_WindowShouldBecomeVisible()
        {
            _testWindow.Hide();
            Assert.IsFalse(_testWindow.IsVisible);

            _testWindow.Show();
            Assert.IsTrue(_testWindow.IsVisible);
        }

        [Test]
        public void Hide_WindowShouldBecomeHidden()
        {
            _testWindow.Show();
            Assert.IsTrue(_testWindow.IsVisible);

            _testWindow.Hide();
            Assert.IsFalse(_testWindow.IsVisible);
        }

        [Test]
        public void SetLayer_WindowShouldHaveCorrectLayer()
        {
            _testWindow.SetLayer(UILayer.Popup);
            Assert.AreEqual(UILayer.Popup, _testWindow.Layer);
        }

        [Test]
        public void OnShown_LifecycleEventShouldFire()
        {
            bool eventFired = false;
            _testWindow.OnShown += () => eventFired = true;

            _testWindow.Show();
            Assert.IsTrue(eventFired);
        }

        [Test]
        public void OnHidden_LifecycleEventShouldFire()
        {
            _testWindow.Show();
            bool eventFired = false;
            _testWindow.OnHidden += () => eventFired = true;

            _testWindow.Hide();
            Assert.IsTrue(eventFired);
        }

        [Test]
        public void OnDestroy_LifecycleEventShouldFire()
        {
            bool eventFired = false;
            _testWindow.OnDestroyed += () => eventFired = true;

            Object.Destroy(_testWindow.gameObject);
            Assert.IsTrue(eventFired);
        }

        private class TestWindow : BaseWindow
        {
            public override string WindowName => "TestWindow";
            protected override void OnShow() { }
            protected override void OnHide() { }
            protected override void OnDestroy() { }
        }
    }

    public class UIManagerTests
    {
        private UIManager _uiManager;
        private TestWindow _testWindow;

        [SetUp]
        public void Setup()
        {
            var gameObject = new GameObject("UIManager");
            _uiManager = gameObject.AddComponent<UIManager>();

            var windowObj = new GameObject("TestWindow");
            _testWindow = windowObj.AddComponent<TestWindow>();
        }

        [TearDown]
        public void Teardown()
        {
            Object.Destroy(_uiManager.gameObject);
            Object.Destroy(_testWindow.gameObject);
        }

        [Test]
        public void Register_WindowShouldBeRegistered()
        {
            _uiManager.RegisterWindow(_testWindow);
            Assert.IsTrue(_uiManager.IsWindowRegistered("TestWindow"));
        }

        [Test]
        public void Unregister_WindowShouldBeUnregistered()
        {
            _uiManager.RegisterWindow(_testWindow);
            _uiManager.UnregisterWindow(_testWindow);
            Assert.IsFalse(_uiManager.IsWindowRegistered("TestWindow"));
        }

        [Test]
        public void FindWindow_RegisteredWindowShouldBeFound()
        {
            _uiManager.RegisterWindow(_testWindow);
            var found = _uiManager.FindWindow("TestWindow");
            Assert.AreEqual(_testWindow, found);
        }

        [Test]
        public void FindWindow_UnregisteredWindowShouldReturnNull()
        {
            var found = _uiManager.FindWindow("NonExistent");
            Assert.IsNull(found);
        }

        [Test]
        public void Show_RegisteredWindowShouldBecomeVisible()
        {
            _uiManager.RegisterWindow(_testWindow);
            _uiManager.ShowWindow("TestWindow");
            Assert.IsTrue(_testWindow.IsVisible);
        }

        [Test]
        public void Hide_WindowShouldBecomeHidden()
        {
            _uiManager.RegisterWindow(_testWindow);
            _uiManager.ShowWindow("TestWindow");
            _uiManager.HideWindow("TestWindow");
            Assert.IsFalse(_testWindow.IsVisible);
        }

        [Test]
        public void ShowStack_WindowsShouldStackCorrectly()
        {
            var window2 = new GameObject("TestWindow2").AddComponent<TestWindow>();
            _uiManager.RegisterWindow(_testWindow);
            _uiManager.RegisterWindow(window2);

            _uiManager.ShowWindow("TestWindow");
            _uiManager.ShowWindow("TestWindow2");

            var stack = _uiManager.GetWindowStack();
            Assert.AreEqual(2, stack.Count);
            Assert.AreEqual("TestWindow2", stack[1].WindowName);

            Object.Destroy(window2.gameObject);
        }

        [Test]
        public void ClearStack_ShouldCloseAllStackedWindows()
        {
            var window2 = new GameObject("TestWindow2").AddComponent<TestWindow>();
            _uiManager.RegisterWindow(_testWindow);
            _uiManager.RegisterWindow(window2);

            _uiManager.ShowWindow("TestWindow");
            _uiManager.ShowWindow("TestWindow2");
            _uiManager.ClearWindowStack();

            Assert.AreEqual(0, _uiManager.GetWindowStack().Count);

            Object.Destroy(window2.gameObject);
        }

        private class TestWindow : BaseWindow

    public class UIEventTests
    {
        private TestWindow _testWindow;

        [SetUp]
        public void Setup()
        {
            var gameObject = new GameObject("TestWindow");
            _testWindow = gameObject.AddComponent<TestWindow>();
        }

        [TearDown]
        public void Teardown()
        {
            Object.Destroy(_testWindow.gameObject);
        }

        [Test]
        public void BindEvent_EventShouldBeTriggered()
        {
            bool triggered = false;
            _testWindow.BindEvent("TestEvent", () => triggered = true);

            _testWindow.TriggerEvent("TestEvent");
            Assert.IsTrue(triggered);
        }

        [Test]
        public void UnbindEvent_EventShouldNotBeTriggered()
        {
            bool triggered = false;
            System.Action handler = () => triggered = true;
            _testWindow.BindEvent("TestEvent", handler);
            _testWindow.UnbindEvent("TestEvent", handler);

            _testWindow.TriggerEvent("TestEvent");
            Assert.IsFalse(triggered);
        }

        [Test]
        public void BubbleEvent_EventShouldBubbleToParent()
        {
            var parentObj = new GameObject("ParentWindow");
            var parentWindow = parentObj.AddComponent<TestWindow>();
            _testWindow.SetParent(parentWindow);

            bool parentTriggered = false;
            parentWindow.BindEvent("TestEvent", () => parentTriggered = true);

            _testWindow.BubbleEvent("TestEvent");
            Assert.IsTrue(parentTriggered);

            Object.Destroy(parentObj);
        }

        private class TestWindow : BaseWindow
        {
            private Dictionary<string, List<System.Action>> _eventHandlers = new Dictionary<string, List<System.Action>>();
            private BaseWindow _parent;

            public override string WindowName => "TestWindow";
            protected override void OnShow() { }
            protected override void OnHide() { }
            protected override void OnDestroy() { }

            public void BindEvent(string eventName, System.Action handler)
            {
                if (!_eventHandlers.ContainsKey(eventName))
                    _eventHandlers[eventName] = new List<System.Action>();
                _eventHandlers[eventName].Add(handler);
            }

            public void UnbindEvent(string eventName, System.Action handler)
            {
                if (_eventHandlers.ContainsKey(eventName))
                    _eventHandlers[eventName].Remove(handler);
            }

            public void TriggerEvent(string eventName)
            {
                if (_eventHandlers.ContainsKey(eventName))
                    foreach (var handler in _eventHandlers[eventName])
                        handler.Invoke();
            }

            public void SetParent(BaseWindow parent)
            {
                _parent = parent;
            }

            public void BubbleEvent(string eventName)
            {
                TriggerEvent(eventName);
                _parent?.BubbleEvent(eventName);
            }
        }
    }

    #region Base Classes

    public enum UILayer
    {
        Root = 0,
        Page = 1,
        Popup = 2,
        Tips = 3
    }

    public abstract class BaseWindow : MonoBehaviour
    {
        public abstract string WindowName { get; }
        public bool IsVisible { get; private set; }
        public UILayer Layer { get; private set; }

        public event System.Action OnShown;
        public event System.Action OnHidden;
        public event System.Action OnDestroyed;

        public void Show()
        {
            if (!IsVisible)
            {
                IsVisible = true;
                OnShow();
                OnShown?.Invoke();
            }
        }

        public void Hide()
        {
            if (IsVisible)
            {
                IsVisible = false;
                OnHide();
                OnHidden?.Invoke();
            }
        }

        public void SetLayer(UILayer layer)
        {
            Layer = layer;
        }

        public void AdjustLayer(UILayer layer)
        {
            SetLayer(layer);
        }

        protected virtual void OnDestroy()
        {
            OnDestroyed?.Invoke();
        }

        protected abstract void OnShow();
        protected abstract void OnHide();
        protected abstract void OnDestroy();
    }

    public class UIManager : MonoBehaviour
    {
        private Dictionary<string, BaseWindow> _registeredWindows = new Dictionary<string, BaseWindow>();
        private List<BaseWindow> _windowStack = new List<BaseWindow>();

        public void RegisterWindow(BaseWindow window)
        {
            if (window == null || string.IsNullOrEmpty(window.WindowName))
                return;

            if (!_registeredWindows.ContainsKey(window.WindowName))
            {
                _registeredWindows.Add(window.WindowName, window);
            }
        }

        public void UnregisterWindow(BaseWindow window)
        {
            if (window == null || string.IsNullOrEmpty(window.WindowName))
                return;

            _registeredWindows.Remove(window.WindowName);
            _windowStack.Remove(window);
        }

        public bool IsWindowRegistered(string windowName)
        {
            return _registeredWindows.ContainsKey(windowName);
        }

        public BaseWindow FindWindow(string windowName)
        {
            return _registeredWindows.TryGetValue(windowName, out var window) ? window : null;
        }

        public void ShowWindow(string windowName)
        {
            var window = FindWindow(windowName);
            if (window != null)
            {
                window.Show();
                if (!_windowStack.Contains(window))
                    _windowStack.Add(window);
            }
        }

        public void HideWindow(string windowName)
        {
            var window = FindWindow(windowName);
            if (window != null)
            {
                window.Hide();
                _windowStack.Remove(window);
            }
        }

        public List<BaseWindow> GetWindowStack()
        {
            return new List<BaseWindow>(_windowStack);
        }

        public void ClearWindowStack()
        {
            foreach (var window in _windowStack.ToArray())
            {
                window.Hide();
            }
            _windowStack.Clear();
        }
    }

    #endregion
}
