## 完整可运行的C#单元测试代码

```csharp
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;
using System.Collections;
using System.Collections.Generic;

namespace GameCore.Tests.ResourceSystem
{
    [TestFixture]
    public class ResourceSystemTests
    {
        private GameObject testPrefab;
        private const string TEST_RESOURCE_PATH = "TestPrefab";

        [SetUp]
        public void SetUp()
        {
            testPrefab = Resources.Load<GameObject>(TEST_RESOURCE_PATH);
        }

        [TearDown]
        public void TearDown()
        {
            Resources.UnloadUnusedAssets();
            System.GC.Collect();
        }

        #region 资源加载测试

        [Test]
        public void TestSyncLoad_加载资源返回非空对象()
        {
            var obj = Resources.Load<GameObject>(TEST_RESOURCE_PATH);
            Assert.IsNotNull(obj, "同步加载资源应该返回非空对象");
        }

        [Test]
        public void TestSyncLoad_加载不存在资源返回空()
        {
            var obj = Resources.Load<GameObject>("NonExistentResource");
            Assert.IsNull(obj, "加载不存在的资源应该返回null");
        }

        [UnityTest]
        public IEnumerator TestAsyncLoad_异步加载返回资源对象()
        {
            ResourceRequest request = Resources.LoadAsync<GameObject>(TEST_RESOURCE_PATH);
            yield return request;

            Assert.IsNotNull(request.asset, "异步加载完成后资源不应为空");
            Assert.AreEqual(typeof(GameObject), request.asset.GetType(), "资源类型应该匹配");
        }

        [UnityTest]
        public IEnumerator TestAsyncLoad_加载进度正确()
        {
            ResourceRequest request = Resources.LoadAsync<GameObject>(TEST_RESOURCE_PATH);
            float lastProgress = 0f;

            while (!request.isDone)
            {
                Assert.GreaterOrEqual(request.progress, lastProgress, "加载进度应该单调递增");
                lastProgress = request.progress;
                yield return null;
            }

            Assert.AreEqual(1f, request.progress, "完成时进度应等于1");
        }

        #endregion

        #region 对象池测试

        [Test]
        public void TestObjectPool_获取对象()
        {
            var poolManager = new GameObject("PoolManager").AddComponent<ObjectPoolManager>();
            var prefab = Resources.Load<GameObject>(TEST_RESOURCE_PATH);

            var obj = poolManager.Get(prefab);
            Assert.IsNotNull(obj, "从对象池获取的对象不应为空");
            Assert.IsTrue(obj.activeSelf, "获取的对象应该是激活状态");

            Object.DestroyImmediate(poolManager.gameObject);
        }

        [Test]
        public void TestObjectPool_归还对象()
        {
            var poolManager = new GameObject("PoolManager").AddComponent<ObjectPoolManager>();
            var prefab = Resources.Load<GameObject>(TEST_RESOURCE_PATH);

            var obj = poolManager.Get(prefab);
            poolManager.Return(obj);

            Assert.IsFalse(obj.activeSelf, "归还后对象应该是非激活状态");

            Object.DestroyImmediate(poolManager.gameObject);
        }

        [Test]
        public void TestObjectPool_对象复用()
        {
            var poolManager = new GameObject("PoolManager").AddComponent<ObjectPoolManager>();
            var prefab = Resources.Load<GameObject>(TEST_RESOURCE_PATH);

            var obj1 = poolManager.Get(prefab);
            poolManager.Return(obj1);

            var obj2 = poolManager.Get(prefab);
            Assert.AreEqual(obj1, obj2, "归还后再次获取应该复用同一对象");

            Object.DestroyImmediate(poolManager.gameObject);
        }

        [Test]
        public void TestObjectPool_容量限制()
        {
            var poolManager = new GameObject("PoolManager").AddComponent<ObjectPoolManager>();
            poolManager.MaxPoolSize = 3;
            var prefab = Resources.Load<GameObject>(TEST_RESOURCE_PATH);

            var objects = new List<GameObject>();
            for (int i = 0; i < 5; i++)
            {
                objects.Add(poolManager.Get(prefab));
            }

            int activeCount = 0;
            foreach (var obj in objects)
            {
                if (obj != null && obj.activeSelf)
                    activeCount++;
            }

            Assert.LessOrEqual(activeCount, poolManager.MaxPoolSize, "活跃对象数不应超过池容量");

            Object.DestroyImmediate(poolManager.gameObject);
        }

        #endregion

        #region 资源卸载测试

        [Test]
        public void TestReferenceCount_引用计数增加()
        {
            var loader = new AssetBundleLoader();
            string testPath = "TestAsset";

            loader.Load(testPath);
            Assert.AreEqual(1, loader.GetReferenceCount(testPath), "首次加载后引用计数应为1");

            loader.Load(testPath);
            Assert.AreEqual(2, loader.GetReferenceCount(testPath), "重复加载后引用计数应为2");
        }

        [Test]
        public void TestReferenceCount_引用计数减少()
        {
            var loader = new AssetBundleLoader();
            string testPath = "TestAsset";

            loader.Load(testPath);
            loader.Load(testPath);
            loader.Unload(testPath);

            Assert.AreEqual(1, loader.GetReferenceCount(testPath), "卸载一次后引用计数应减1");
        }

        [Test]
        public void TestUnload_资源释放()
        {
            var loader = new AssetBundleLoader();
            string testPath = "TestAsset";

            loader.Load(testPath);
            loader.Unload(testPath);

            Assert.AreEqual(0, loader.GetReferenceCount(testPath), "完全卸载后引用计数应为0");
            Assert.IsFalse(loader.IsLoaded(testPath), "完全卸载后资源应处于未加载状态");
        }

        [Test]
        public void TestForceUnload_强制卸载()
        {
            var loader = new AssetBundleLoader();
            string testPath = "TestAsset";

            loader.Load(testPath);
            loader.Load(testPath);
            loader.ForceUnload(testPath);

            Assert.AreEqual(0, loader.GetReferenceCount(testPath), "强制卸载后引用计数应为0");
            Assert.IsFalse(loader.IsLoaded(testPath), "强制卸载后资源应处于未加载状态");
        }

        #endregion

        #region 资源查找测试

        [Test]
        public void TestResourceLocation_资源定位()
        {
            var resourceInfo = ResourceManager.Locate(TEST_RESOURCE_PATH);
            Assert.IsNotNull(resourceInfo, "资源定位应返回非空结果");
            Assert.AreEqual(TEST_RESOURCE_PATH, resourceInfo.Path, "资源路径应匹配");
        }

        [Test]
        public void TestResourceCache_缓存命中()
        {
            var resourceManager = ResourceManager.Instance;

            var resource1 = resourceManager.Load<GameObject>(TEST_RESOURCE_PATH);
            var resource2 = resourceManager.Load<GameObject>(TEST_RESOURCE_PATH);

            Assert.AreEqual(resource1, resource2, "缓存的资源应该是同一引用");
        }

        [Test]
        public void TestResourceCache_缓存未命中()
        {
            var resourceManager = ResourceManager.Instance;
            resourceManager.UnloadAll();

            var resource = resourceManager.Load<GameObject>("DifferentResource");
            Assert.IsNull(resource, "不存在的资源加载应返回空");
        }

        #endregion
    }

    #region Mock Classes

    public class ObjectPoolManager : MonoBehaviour
    {
        private Dictionary<GameObject, Queue<GameObject>> poolDict = new Dictionary<GameObject, Queue<GameObject>>();
        public int MaxPoolSize = 10;

        public GameObject Get(GameObject prefab)
        {
            if (poolDict.ContainsKey(prefab) && poolDict[prefab].Count > 0)
            {
                var obj = poolDict[prefab].Dequeue();
                obj.SetActive(true);
                return obj;
            }

            return Instantiate(prefab);
        }

        public void Return(GameObject obj)
        {
            obj.SetActive(false);
            var prefab = obj.GetComponent<ObjectPoolItem>()?.Prefab;

            if (prefab != null)
            {
                if (!poolDict.ContainsKey(prefab))
                    poolDict[prefab] = new Queue<GameObject>();

                if (poolDict[prefab].Count < MaxPoolSize)
                    poolDict[prefab].Enqueue(obj);
                else
                    Destroy(obj);
            }
        }
    }

    public class ObjectPoolItem : MonoBehaviour
    {
        public GameObject Prefab;
    }

    public class AssetBundleLoader
    {
        private Dictionary<string, int> referenceCount = new Dictionary<string, int>();
        private HashSet<string> loadedAssets = new HashSet<string>();

        public void Load(string path)
        {
            if (!referenceCount.ContainsKey(path))
                referenceCount[path] = 0;

            referenceCount[path]++;
            loadedAssets.Add(path);
        }

        public void Unload(string path)
        {
            if (referenceCount.ContainsKey(path) && referenceCount[path] > 0)
                referenceCount[path]--;
        }

        public void ForceUnload(string path)
        {
            referenceCount[path] = 0;
            loadedAssets.Remove(path);
        }

        public int GetReferenceCount(string path)
        {
            return referenceCount.ContainsKey(path) ? referenceCount[path] : 0;
        }

        public bool IsLoaded(string path)
        {
            return loadedAssets.Contains(path) && GetReferenceCount(path) > 0;
        }
    }

    public class ResourceManager
    {
        private static ResourceManager instance;
        private Dictionary<string, Object> cache = new Dictionary<string, Object>();

        public static ResourceManager Instance
        {
            get
            {
                if (instance == null)
                    instance = new ResourceManager();
                return instance;
            }
        }

        public T Load<T>(string path) where T : Object
        {
            if (cache.ContainsKey(path))
                return cache[path] as T;

            var resource = Resources.Load<T>(path);
            if (resource != null)
                cache[path] = resource;

            return resource;
        }

        public void UnloadAll()
        {
            cache.Clear();
            Resources.UnloadUnusedAssets();
        }

        public static ResourceInfo Locate(string path)
        {
            return new ResourceInfo { Path = path };
        }
    }

    public class ResourceInfo
    {
        public string Path;
        public string FullPath;
        public Object Asset;
    }

    #endregion
}
```

## 测试执行说明

1. 将测试文件放入 `Assets/Tests/` 目录
2. 使用 Unity Test Runner 运行测试
3. 确保 `TestPrefab` 资源存在于 `Resources` 目录下
4. 所有测试用例可通过 Unity 的 Test Runner 批量执行
