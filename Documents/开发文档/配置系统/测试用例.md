# 配置系统单元测试

## 测试环境
使用 NUnit 和 Unity Test Framework

## 测试用例

### ConfigManager测试
- 测试配置管理器初始化
- 测试配置加载
- 测试配置获取
- 测试配置缓存

### 配置表数据测试
- 测试MapDataConfig数据完整性
- 测试TileDataConfig数据完整性
- 测试配置数据解析

### ScriptableObject测试
- 测试SO创建
- 测试SO数据读取
- 测试SO数据修改

### 代码生成测试
- 测试配置类属性访问
- 测试数据类型转换

## 完整可运行的C#单元测试代码

```csharp
using NUnit.Framework;
using UnityEngine;
using System.Collections.Generic;
using System.IO;

namespace CardRougelike.Config.Tests
{
    public class ConfigTestBase
    {
        protected ConfigManager configManager;
        
        [OneTimeSetUp]
        public void GlobalSetup()
        {
            if (!ConfigManager.IsInitialized)
            {
                var go = new GameObject("ConfigManager");
                configManager = go.AddComponent<ConfigManager>();
            }
            else
            {
                configManager = ConfigManager.Instance;
            }
        }
        
        [OneTimeTearDown]
        public void GlobalTeardown()
        {
            if (configManager != null)
            {
                Object.DestroyImmediate(configManager.gameObject);
            }
        }
    }
    
    public class ConfigManagerTests : ConfigTestBase
    {
        [Test]
        public void TestConfigManagerInitialization()
        {
            Assert.IsNotNull(ConfigManager.Instance, "ConfigManager单例应该已初始化");
            Assert.IsTrue(ConfigManager.IsInitialized, "ConfigManager应该已标记为已初始化");
        }
        
        [Test]
        public void TestConfigLoading()
        {
            bool loadComplete = false;
            ConfigManager.Instance.OnConfigLoadComplete += () => loadComplete = true;
            
            ConfigManager.Instance.LoadAllConfigs();
            
            Assert.IsTrue(loadComplete, "配置加载应该完成");
        }
        
        [Test]
        public void TestConfigCaching()
        {
            var config1 = ConfigManager.Instance.GetConfig<MapDataConfig>();
            var config2 = ConfigManager.Instance.GetConfig<MapDataConfig>();
            
            Assert.AreSame(config1, config2, "相同配置的多次获取应该返回相同实例");
        }
    }
    
    public class MapDataConfigTests : ConfigTestBase
    {
        [Test]
        public void TestMapDataConfigExists()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
        }
        
        [Test]
        public void TestMapDataConfigDataIntegrity()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig不应该为空");
            
            var mapList = mapConfig.GetAll();
            Assert.IsNotNull(mapList, "地图配置列表不应该为空");
        }
        
        [Test]
        public void TestMapDataConfigById()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
            
            var mapList = mapConfig.GetAll();
            if (mapList != null && mapList.Count > 0)
            {
                var firstMap = mapList[0];
                var retrievedMap = mapConfig.Get(firstMap.Id);
                Assert.AreEqual(firstMap.Id, retrievedMap.Id, "通过ID获取的地图数据应该匹配");
            }
        }
    }
    
    public class TileDataConfigTests : ConfigTestBase
    {
        [Test]
        public void TestTileDataConfigExists()
        {
            var tileConfig = ConfigManager.Instance.GetConfig<TileDataConfig>();
            Assert.IsNotNull(tileConfig, "TileDataConfig应该存在");
        }
        
        [Test]
        public void TestTileDataConfigDataIntegrity()
        {
            var tileConfig = ConfigManager.Instance.GetConfig<TileDataConfig>();
            Assert.IsNotNull(tileConfig, "TileDataConfig不应该为空");
            
            var tileList = tileConfig.GetAll();
            Assert.IsNotNull(tileList, "瓦片配置列表不应该为空");
        }
    }
    
    public class ConfigDataParsingTests : ConfigTestBase
    {
        [Test]
        public void TestConfigDataParsing_IntField()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
            
            var mapList = mapConfig.GetAll();
            if (mapList != null && mapList.Count > 0)
            {
                Assert.GreaterOrEqual(mapList[0].Id, 0, "地图ID应该是非负整数");
            }
        }
        
        [Test]
        public void TestConfigDataParsing_StringField()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
            
            var mapList = mapConfig.GetAll();
            if (mapList != null && mapList.Count > 0)
            {
                Assert.IsNotNull(mapList[0].Name, "地图名称不应该为空");
            }
        }
    }
    
    public class ScriptableObjectTests
    {
        [Test]
        public void TestSOCreation()
        {
            var scriptableObject = ScriptableObject.CreateInstance<MapDataConfig>();
            Assert.IsNotNull(scriptableObject, "ScriptableObject应该能成功创建");
            
            Object.DestroyImmediate(scriptableObject);
        }
        
        [Test]
        public void TestSODataReading()
        {
            var scriptableObject = ScriptableObject.CreateInstance<MapDataConfig>();
            Assert.IsNotNull(scriptableObject, "ScriptableObject应该存在");
            
            Object.DestroyImmediate(scriptableObject);
        }
        
        [Test]
        public void TestSODataModification()
        {
            var scriptableObject = ScriptableObject.CreateInstance<MapDataConfig>();
            Assert.IsNotNull(scriptableObject, "ScriptableObject应该存在");
            
            Object.DestroyImmediate(scriptableObject);
        }
    }
    
    public class ConfigCodeGenerationTests : ConfigTestBase
    {
        [Test]
        public void TestConfigClassPropertyAccess()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
            
            var mapList = mapConfig.GetAll();
            Assert.IsNotNull(mapList, "配置列表应该可以正常访问");
        }
        
        [Test]
        public void TestDataTypeConversion()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
            
            var mapList = mapConfig.GetAll();
            if (mapList != null && mapList.Count > 0)
            {
                int id = mapList[0].Id;
                string name = mapList[0].Name;
                
                Assert.AreEqual(id.ToString(), id.ToString(), "整数类型转换应该正常");
                Assert.AreEqual(name, name, "字符串类型应该保持一致");
            }
        }
        
        [Test]
        public void TestConfigListIteration()
        {
            var mapConfig = ConfigManager.Instance.GetConfig<MapDataConfig>();
            Assert.IsNotNull(mapConfig, "MapDataConfig应该存在");
            
            var mapList = mapConfig.GetAll();
            Assert.IsNotNull(mapList, "配置列表应该可以迭代");
            
            int count = 0;
            foreach (var map in mapList)
            {
                Assert.IsNotNull(map, "列表中的每个元素都不应该为空");
                count++;
            }
            
            Assert.GreaterOrEqual(count, 0, "列表迭代应该正常完成");
        }
    }
}
```
