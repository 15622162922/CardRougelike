        [Test]
        public void TestMapGenerationWithSameSeed()

            _config.seed = 12345;
            var map2 = _generator.Generate(_config);

            Assert.AreEqual(map1.seed, map2.seed);
            Assert.AreEqual(map1.rooms.Count, map2.rooms.Count);
        }

        [Test]
        public void TestMapGenerationCreatesStartRoom()
        {
            var map = _generator.Generate(_config);

            var startRoom = map.rooms.FirstOrDefault(r => r.roomType == RoomType.Start);
            Assert.IsNotNull(startRoom, "Map should have a start room");
        }

        [Test]
        public void TestMapGenerationCreatesBossRoom()
        {
            var map = _generator.Generate(_config);

            var bossRoom = map.rooms.FirstOrDefault(r => r.roomType == RoomType.Boss);
            Assert.IsNotNull(bossRoom, "Map should have a boss room");
        }

        [Test]
        public void TestMapGenerationCreatesConnections()
        {
            var map = _generator.Generate(_config);

            Assert.IsNotNull(map.connections);
            Assert.Greater(map.connections.Count, 0, "Map should have at least one connection");
        }

        [Test]
        public void TestAllRoomsAreConnected()
        {
            var map = _generator.Generate(_config);

            var visited = new HashSet<string>();
            var queue = new Queue<string>();
            queue.Enqueue(map.startRoomId);
            visited.Add(map.startRoomId);

            while (queue.Count > 0)
            {
                var currentId = queue.Dequeue();
                var connections = map.connections.Where(c => c.fromRoomId == currentId || c.toRoomId == currentId);

                foreach (var conn in connections)
                {
                    var neighborId = conn.fromRoomId == currentId ? conn.toRoomId : conn.fromRoomId;
                    if (!visited.Contains(neighborId))
                    {
                        visited.Add(neighborId);
                        queue.Enqueue(neighborId);
                    }
                }
            }

            Assert.AreEqual(map.rooms.Count, visited.Count, "All rooms should be connected");
        }

        [Test]
        public void TestBossRoomIsReachable()
        {
            var map = _generator.Generate(_config);

            var visited = new HashSet<string>();
            var queue = new Queue<string>();
            queue.Enqueue(map.startRoomId);
            visited.Add(map.startRoomId);

            while (queue.Count > 0)
            {
                var currentId = queue.Dequeue();
                var connections = map.connections.Where(c => c.fromRoomId == currentId || c.toRoomId == currentId);

                foreach (var conn in connections)
                {
                    var neighborId = conn.fromRoomId == currentId ? conn.toRoomId : conn.fromRoomId;
                    if (!visited.Contains(neighborId))
                    {
                        visited.Add(neighborId);
                        queue.Enqueue(neighborId);
                    }
                }
            }

            Assert.IsTrue(visited.Contains(map.bossRoomId), "Boss room should be reachable from start room");
        }

        [Test]
        public void TestRoomCountWithinRange()
        {
            var map = _generator.Generate(_config);

            Assert.GreaterOrEqual(map.rooms.Count, _config.minPathLength);
            Assert.LessOrEqual(map.rooms.Count, _config.maxPathLength + _config.branchCount * 3);
        }
    }

    public static class RandomChunkLayout
    {
        public static ChunkLayout Generate(int seed, int width, int height)
        {
            var random = new System.Random(seed);
            var layout = new ChunkLayout
            {
                Width = width,
                Height = height,
                Chunks = new List<ChunkData>()
            };

            for (int i = 0; i < 4; i++)
            {
                layout.Chunks.Add(new ChunkData
                {
                    ChunkId = i,
                    OffsetX = random.Next(width),
                    OffsetY = random.Next(height)
                });
            }

            return layout;
        }
    }

    public class ChunkLayout
    {
        public int Width { get; set; }
        public int Height { get; set; }
        public List<ChunkData> Chunks { get; set; }
    }

    public class ChunkData
    {
        public int ChunkId { get; set; }
        public int OffsetX { get; set; }
        public int OffsetY { get; set; }
    }

    public class MapModule
    {
        public event System.EventHandler<MapChangeEventArgs> OnMapChanged;
        public int CurrentMapId { get; private set; }

        public void SwitchMap(int newMapId)
        {
            var oldId = CurrentMapId;
            CurrentMapId = newMapId;
            NotifyMapChanged(oldId, newMapId);
        }

        public void NotifyMapChanged(int oldMapId, int newMapId)
        {
            OnMapChanged?.Invoke(this, new MapChangeEventArgs
            {
                OldMapId = oldMapId,
                NewMapId = newMapId
            });
        }

        public MapData GetMapData(int mapId)
        {
            return new MapData
            {
                MapId = mapId,
                MapName = $"Map_{mapId}",
                Width = 10,
                Height = 10,
                Seed = mapId * 1000
            };
        }
    }

    public class MapManager : MonoBehaviour
    {
        public List<ChunkData> GenerateChunks(int width, int height, int seed)
        {
            var layout = RandomChunkLayout.Generate(seed, width, height);
            return layout.Chunks;
        }

        public (Vector3Int start, Vector3Int end) GenerateStartAndEndPoints(int width, int height)
        {
            var start = new Vector3Int(0, 0, 0);
            var end = new Vector3Int(width - 1, height - 1, 0);
            return (start, end);
        }
    }

    public class RoomGraphGenerator
    {
        private MapGenerationConfig _config;
        private System.Random _random;
        private List<GeneratedRoom> _rooms;
        private List<RoomConnection> _connections;
        private int _roomCounter;
        private List<List<GeneratedRoom>> _mainPaths;

        public GeneratedMap Generate(MapGenerationConfig config)
        {
            _config = config;
            _random = new System.Random(config.seed);
            _rooms = new List<GeneratedRoom>();
            _connections = new List<RoomConnection>();
            _mainPaths = new List<List<GeneratedRoom>>();
            _roomCounter = 0;

            var startRoom = CreateStartRoom();
            var bossRoom = CreateBossRoom();
            _rooms.Add(startRoom);
            _rooms.Add(bossRoom);

            _mainPaths = GenerateMainPaths(startRoom, bossRoom);

            foreach (var path in _mainPaths)
            {
                AddBranchesToPath(path);
            }

            ConnectPathsHorizontally();

            return new GeneratedMap
            {
                seed = config.seed,
                rooms = _rooms,
                connections = _connections,
                startRoomId = startRoom.instanceId,
                bossRoomId = bossRoom.instanceId
            };
        }

        private GeneratedRoom CreateStartRoom()
        {
            return new GeneratedRoom
            {
                instanceId = $"room_{_roomCounter++}",
                templateId = _config.startRoomTemplate?.roomId ?? "start_room",
                roomType = RoomType.Start,
                gridPosition = Vector2Int.zero,
                rotation = Quaternion.identity,
                exitIds = new List<string> { $"exit_{_roomCounter}" }
            };
        }

        private GeneratedRoom CreateBossRoom()
        {
            return new GeneratedRoom
            {
                instanceId = $"room_{_roomCounter++}",
                templateId = _config.bossRoomTemplate?.roomId ?? "boss_room",
                roomType = RoomType.Boss,
                gridPosition = new Vector2Int(0, _config.maxPathLength + 1),
                rotation = Quaternion.identity,
                exitIds = new List<string> { $"exit_{_roomCounter}" }
            };
        }

        private List<List<GeneratedRoom>> GenerateMainPaths(GeneratedRoom start, GeneratedRoom end)
        {
            var paths = new List<List<GeneratedRoom>>();
            var pathCount = _random.Next(2, _config.branchCount + 1);

            for (int p = 0; p < pathCount; p++)
            {
                var path = new List<GeneratedRoom> { start };
                var pathLength = _random.Next(_config.minPathLength, _config.maxPathLength + 1);

                for (int i = 1; i <= pathLength; i++)
                {
                    var yOffset = (end.gridPosition.y - start.gridPosition.y) * i / (pathLength + 1);
                    var xOffset = p * 2;

                    var newRoom = CreatePathRoom(new Vector2Int(xOffset, start.gridPosition.y + yOffset), RoomType.Normal);
                    path.Add(newRoom);
                }

                path.Add(end);

                for (int i = 0; i < path.Count - 1; i++)
                {
                    CreateConnection(path[i], path[i + 1], ConnectionType.Normal);
                }

                paths.Add(path);
            }

            return paths;
        }

        private void AddBranchesToPath(List<GeneratedRoom> path)
        {
            for (int i = 1; i < path.Count - 1; i++)
            {
                var room = path[i];
                var branchChance = _random.Next(100);

                if (branchChance < 30)
                {
                    var branchRoom = CreatePathRoom(
                        new Vector2Int(room.gridPosition.x + _random.Next(-2, 3), room.gridPosition.y),
                        GetRandomRoomType()
                    );

                    CreateConnection(room, branchRoom, ConnectionType.Corridor);
                }
            }
        }

        private void ConnectPathsHorizontally()
        {
            if (_mainPaths.Count < 2) return;

            for (int i = 0; i < _mainPaths.Count - 1; i++)
            {
                var path1 = _mainPaths[i];
                var path2 = _mainPaths[i + 1];

                var connectIndex1 = _random.Next(1, path1.Count - 1);
                var connectIndex2 = _random.Next(1, path2.Count - 1);

                CreateConnection(path1[connectIndex1], path2[connectIndex2], ConnectionType.Door);
            }
        }

        private GeneratedRoom CreatePathRoom(Vector2Int position, RoomType type)
        {
            var room = new GeneratedRoom
            {
                instanceId = $"room_{_roomCounter++}",
                templateId = GetTemplateIdForType(type),
                roomType = type,
                gridPosition = position,
                rotation = Quaternion.identity,
                exitIds = new List<string> { $"exit_{_roomCounter}" }
            };
            _rooms.Add(room);
            return room;
        }

        private void CreateConnection(GeneratedRoom from, GeneratedRoom to, ConnectionType type)
        {
            _connections.Add(new RoomConnection
            {
                fromRoomId = from.instanceId,
                fromExitId = from.exitIds[0],
                toRoomId = to.instanceId,
                toExitId = to.exitIds[0],
                connectionType = type
            });
        }

        private RoomType GetRandomRoomType()
        {
            var types = new[] { RoomType.Normal, RoomType.Normal, RoomType.Elite, RoomType.Rest };
            return types[_random.Next(types.Length)];
        }

        private string GetTemplateIdForType(RoomType type)
        {
            return type switch
            {
                RoomType.Elite => "elite_room",
                RoomType.Rest => "rest_room",
                RoomType.Shop => "shop_room",
                RoomType.Treasure => "treasure_room",
                _ => "normal_room"
            };
        }
    }

    public class RoomPlacer
    {
        private const int CELL_SIZE = 20;
        private Dictionary<Vector2Int, GeneratedRoom> _occupiedCells;

        public void PlaceRooms(GeneratedMap map)
        {
            _occupiedCells = new Dictionary<Vector2Int, GeneratedRoom>();
            var startRoom = map.rooms.First(r => r.roomType == RoomType.Start);

            PlaceRoom(startRoom, Vector2Int.zero);
            PlaceRemainingRooms(map);
        }

        private void PlaceRoom(GeneratedRoom room, Vector2Int gridPos)
        {
            room.gridPosition = gridPos;
            _occupiedCells[gridPos] = room;

            for (int x = -1; x <= 1; x++)
            {
                for (int y = -1; y <= 1; y++)
                {
                    _occupiedCells[new Vector2Int(gridPos.x + x, gridPos.y + y)] = room;
                }
            }
        }

        private void PlaceRemainingRooms(GeneratedMap map)
        {
            foreach (var connection in map.connections)
            {
                var fromRoom = map.rooms.Find(r => r.instanceId == connection.fromRoomId);
                var toRoom = map.rooms.Find(r => r.instanceId == connection.toRoomId);

                if (toRoom.gridPosition == Vector2Int.zero)
                {
                    var newPos = CalculateNextPosition(fromRoom, connection);
                    PlaceRoom(toRoom, newPos);
                }
            }
        }

        private Vector2Int CalculateNextPosition(GeneratedRoom fromRoom, RoomConnection connection)
        {
            return new Vector2Int(fromRoom.gridPosition.x, fromRoom.gridPosition.y + 1);
        }
    }

    public class ExitEditor
    {
        private RoomTemplate _template;

        public void Initialize(RoomTemplate template)
        {
            _template = template;
        }

        public void CreateExit(Vector2Int position, ExitDirection direction)
        {
            var exit = new ExitData
            {
                exitId = System.Guid.NewGuid().ToString(),
                tilePosition = position,
                direction = direction,
                exitType = ExitType.Normal,
                isOpen = true
            };

            _template.exits.Add(exit);
        }

        public void ConnectExits(ExitData from, ExitData to)
        {
            from.connectedExitId = to.exitId;
            to.connectedExitId = from.exitId;
        }
    }

    public class TilePainter
    {
        private Dictionary<Vector2Int, TilePaintData> _placedTiles;
        private RoomTemplate _currentTemplate;

        public void Initialize(RoomTemplate template)
        {
            _currentTemplate = template;
            _placedTiles = new Dictionary<Vector2Int, TilePaintData>();
        }

        public void PaintTile(Vector2Int position, Tile tile)
        {
            _placedTiles[position] = new TilePaintData { tile = tile, position = position };
        }

        public void EraseTile(Vector2Int position)
        {
            _placedTiles.Remove(position);
        }

        public void RotateTile(Vector2Int position)
        {
            if (_placedTiles.TryGetValue(position, out var data))
            {
                data.rotation = (data.rotation + 90) % 360;
            }
        }

        public TilePaintData GetTileAt(Vector2Int position)
        {
            return _placedTiles.TryGetValue(position, out var data) ? data : null;
        }

        public int GetPlacedTileCount()
        {
            return _placedTiles.Count;
        }
    }

    public class TilePaintData
    {
        public Tile tile;
        public Vector2Int position;
        public int rotation;
    }
